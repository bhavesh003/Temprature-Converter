report.component.ts (NO PACKAGE VERSION)
TypeScript
import {
  Component,
  OnInit,
  AfterViewInit
} from '@angular/core';

declare var Chart: any;

interface StatCard {
  title: string;
  value: number;
  icon: string;
  trend: string;
  gradient: string;
}

@Component({
  selector: 'app-report',
  templateUrl: './report.component.html',
  styleUrls: ['./report.component.scss']
})
export class ReportComponent
  implements OnInit, AfterViewInit {

  statCards: StatCard[] = [];

  freelancers = [
    {
      username: 'John Doe',
      email: 'john@gmail.com',
      skills: ['Angular', 'Spring Boot'],
      status: 'Elite'
    },
    {
      username: 'Sarah Smith',
      email: 'sarah@gmail.com',
      skills: ['UI/UX', 'Figma'],
      status: 'Pro'
    }
  ];

  activities = [
    {
      title: 'New proposal submitted',
      time: '2 mins ago',
      icon: 'fa-file-circle-plus',
      color: 'primary'
    },
    {
      title: 'Job approved',
      time: '10 mins ago',
      icon: 'fa-circle-check',
      color: 'success'
    }
  ];

  ngOnInit(): void {
    this.loadStats();
  }

  ngAfterViewInit(): void {
    this.renderCharts();
  }

  loadStats(): void {

    this.statCards = [
      {
        title: 'Total Clients',
        value: 1240,
        icon: 'fa-users',
        trend: '+12%',
        gradient: 'gradient-indigo'
      },
      {
        title: 'Freelancers',
        value: 3280,
        icon: 'fa-user-tie',
        trend: '+18%',
        gradient: 'gradient-purple'
      },
      {
        title: 'Jobs',
        value: 850,
        icon: 'fa-briefcase',
        trend: '+8%',
        gradient: 'gradient-cyan'
      },
      {
        title: 'Proposals',
        value: 5200,
        icon: 'fa-file-signature',
        trend: '+22%',
        gradient: 'gradient-pink'
      }
    ];

  }

  renderCharts(): void {

    // PIE CHART

    new Chart('pieChart', {
      type: 'doughnut',
      data: {
        labels: ['Clients', 'Freelancers'],
        datasets: [{
          data: [1240, 3280],
          backgroundColor: [
            '#6366f1',
            '#06b6d4'
          ]
        }]
      },
      options: {
        responsive: true,
        plugins: {
          legend: {
            labels: {
              color: '#fff'
            }
          }
        }
      }
    });

    // BAR CHART

    new Chart('barChart', {
      type: 'bar',
      data: {
        labels: [
          'Jan',
          'Feb',
          'Mar',
          'Apr',
          'May',
          'Jun'
        ],
        datasets: [{
          label: 'Jobs Posted',
          data: [10, 20, 15, 35, 28, 50],
          borderRadius: 12,
          backgroundColor: '#8b5cf6'
        }]
      },
      options: {
        responsive: true,
        plugins: {
          legend: {
            labels: {
              color: '#fff'
            }
          }
        },
        scales: {
          y: {
            ticks: {
              color: '#fff'
            }
          },
          x: {
            ticks: {
              color: '#fff'
            }
          }
        }
      }
    });

    // LINE CHART

    new Chart('lineChart', {
      type: 'line',
      data: {
        labels: [
          'Jan',
          'Feb',
          'Mar',
          'Apr',
          'May',
          'Jun'
        ],
        datasets: [{
          label: 'Growth',
          data: [100, 300, 600, 900, 1400, 2200],
          borderColor: '#06b6d4',
          backgroundColor: 'rgba(6,182,212,0.2)',
          fill: true,
          tension: 0.4
        }]
      },
      options: {
        responsive: true,
        plugins: {
          legend: {
            labels: {
              color: '#fff'
            }
          }
        },
        scales: {
          y: {
            ticks: {
              color: '#fff'
            }
          },
          x: {
            ticks: {
              color: '#fff'
            }
          }
        }
      }
    });

  }

}
